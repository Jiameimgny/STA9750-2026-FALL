“All electricity generation in a state is fungible. We will assign the average generation emissions to all electric transit in a state.” We’ll parse the EIA SEP summary pages and create a “tidy” table for this analysis.

get_eia_sep <- function(state, abbr){
    library(tidyverse)
    library(stringr)
    library(httr2)
    library(rvest)
    library(fs)
  
    state_formatted <- str_to_lower(state) |> str_replace_all("\\s", "")
    
    dir_name <- fs::path("data", "mp02")
    file_name <- fs::path(dir_name, state_formatted)
    
    fs::dir_create(dir_name, recurse=TRUE)
    
    if(!fs::file_exists(file_name)){
        BASE_URL <- "https://www.eia.gov"
        RESPONSE <- request(BASE_URL) |> 
          req_url_path("electricity", 
                       "state", 
                       state_formatted) |>
          req_headers(
            `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0"
            ) |>   
          req_retry(
                max_tries = 4, 
          ) |>
          req_perform(file_name)
    
        resp_check_status(RESPONSE)
    }
    
    TABLE <- read_html(file_name) |> 
        html_element("table") |> 
        html_table() |>
        mutate(Item = str_to_lower(Item))
    
    if("U.S. rank" %in% colnames(TABLE)){
        TABLE <- TABLE |> rename(Rank = `U.S. rank`)
    }
    
    CO2_MWh <- TABLE |> 
        filter(Item == "carbon dioxide (lbs/mwh)") |>
        pull(Value) |> 
        str_replace_all(",", "") |>
        as.numeric()
    
    PRIMARY <- TABLE |> 
        filter(Item == "primary energy source") |> 
        pull(Rank)
    
    RATE <- TABLE |>
        filter(Item == "average retail price (cents/kwh)") |>
        pull(Value) |>
        as.numeric()
    
    GENERATION_MWh <- TABLE |>
        filter(Item == "net generation (megawatthours)") |>
        pull(Value) |>
        str_replace_all(",", "") |>
        as.numeric()
    
    data.frame(CO2_MWh               = CO2_MWh, 
               primary_source        = PRIMARY,
               electricity_price_MWh = RATE * 10, # / 100 cents to dollars &
               # * 1000 kWh to MWH 
               generation_MWh        = GENERATION_MWh, 
               state                 = state, 
               abbreviation          = abbr
    )
}

EIA_SEP_REPORT <- map2(state.name, state.abb, get_eia_sep) |> list_rbind()

load_ntd_energy <- function(){
  library(readxl)
  library(httr2)
  library(tidyverse)
  library(fs)
  
  # Create 'data/mp02' directory if not already present
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse=TRUE)

  NTD_ENERGY_FILE <- fs::path(DATA_DIR, "2025_ntd_energy.xlsx")

  if(!fs::file_exists(NTD_ENERGY_FILE)){
    library(httr2)
  
    request("https://www.transit.dot.gov/") |>
      req_url_path("sites",
                   "fta.dot.gov",
                   "files",
                   "2026-09",
                   "2025 Energy Consumption_260806.xlsx") |> 
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0") |> 
      req_perform(NTD_ENERGY_FILE)
  }
  
  if(file.info(NTD_ENERGY_FILE)$size < 1e4){
        cat("I was unable to download the NTD Energy File. Please try again.\n")
        stop("Download failed. Please delete the corrupted file by running `unlink('data/mp02/2025_ntd_energy.xlsx')` and try again.")
    }

  to_numeric_fill_0 <- function(x){
    x <- if_else(x == "-", NA, x)
    replace_na(as.numeric(x), 0)
  }

  read_excel(NTD_ENERGY_FILE) |> 
    select(-c(`Reporter Type`, 
              `Reporting Module`, 
              `Gallons of Other Fuel`, 
              `Other Fuel Description`)) |>
    mutate(across(-c(`Agency Name`, 
                     `Mode`,
                     `TOS`), 
                  to_numeric_fill_0)) |>
    group_by(`NTD ID`, `Mode`, `Agency Name`) |>
    summarize(across(where(is.numeric), sum), 
              .groups = "keep") |>
    mutate(ENERGY = sum(c_across(c(where(is.numeric))))) |>
    filter(ENERGY > 0) |>
    select(-ENERGY) |>
    ungroup()
}

NTD_ENERGY <- load_ntd_energy()

load_ntd_service <- function(){
  library(httr2)
  library(tidyverse)
  library(fs)
  
  # Create 'data/mp02' directory if not already present
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse=TRUE)

  NTD_SERVICE_FILE <- fs::path(DATA_DIR, "2025_ntd_service.csv")

  if(!fs::file_exists(NTD_SERVICE_FILE)){
    library(httr2)
  
    request("https://data.transportation.gov/resource/4fir-qbim.csv") |>
      req_url_query(`$query`=paste(
        "SELECT  `_5_digit_ntd_id` AS `NTD_ID`", 
        "`max_agency` AS `AGENCY`",
        "`max_reporter_type` AS `reporter_type`",
        "`max_organization_type` AS `organization_type`",
        "`max_city` AS `city`",
        "`max_state` AS `state`",
        "`mode`",
        "`type_of_service`",
        "`max_primary_uza_code` as `uza_code`",
        "`max_primary_uza_name` as `uza_name`",
        "`max_primary_uza_area_sq_miles` as `uza_sq_miles`",
        "`max_primary_uza_population` as `uza_population`",
        "`max_service_area_sq_miles` as `service_area_sq_miles`",
        "`max_service_area_population` as `service_area_population`",
        "`questionable_record` as `flagged_likely_inaccurate`",
        "`sum_unlinked_passenger_trips_upt` as `total_upt`",
        "`sum_passenger_miles` WHERE caseless_one_of(`report_year`, '2025') limit 50000", 
        sep=",")) |>
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0") |> 
      req_perform(NTD_SERVICE_FILE)
  }
  
  if(file.info(NTD_SERVICE_FILE)$size < 1e4){
        cat("I was unable to download the NTD Service File. Please try again.\n")
        stop("Download failed. Please delete the corrupted file by running `unlink('data/mp02/2025_ntd_service.csv')` and try again.")
    }

  read_csv(NTD_SERVICE_FILE) |>
    mutate(flagged_likely_inaccurate = !is.na(flagged_likely_inaccurate))
}

# 1. Calculate fuel and electricity totals from NTD_ENERGY
energy_cols_df <- NTD_ENERGY |>
  mutate(
    gas_total = select(cur_data(), any_of(matches("Gasoline", ignore.case = TRUE))) |> rowSums(na.rm = TRUE),
    diesel_total = select(cur_data(), any_of(matches("Diesel", ignore.case = TRUE))) |> rowSums(na.rm = TRUE),
    other_fuel_total = select(cur_data(), any_of(matches("LPG|LNG|Compressed Natural Gas|Propane", ignore.case = TRUE))) |> rowSums(na.rm = TRUE),
    
    fuel_emissions = (gas_total * 8.78) + (diesel_total * 10.21) + (other_fuel_total * 6.5),
    
    total_kwh = select(cur_data(), any_of(matches("Kilowatt hours|Electric", ignore.case = TRUE))) |> rowSums(na.rm = TRUE),
    electric_mwh = total_kwh / 1000
  ) |>
  select(any_of(c("NTD ID", "Agency Name", "Mode", "fuel_emissions", "electric_mwh")))

# 2. Get State CO2 intensity for electric transit
agency_state_co2 <- NTD_SERVICE |>
  distinct(NTD_ID, state) |>
  mutate(NTD_ID = as.numeric(NTD_ID)) |>
  left_join(EIA_SEP_REPORT |> select(abbreviation, CO2_MWh), by = c("state" = "abbreviation"))

# 3. Calculate total actual emissions vs baseline driving emissions
transit_summary <- NTD_SERVICE |>
  mutate(NTD_ID = as.numeric(NTD_ID)) |>
  left_join(energy_cols_df, by = c("NTD_ID" = "NTD ID")) |>
  left_join(agency_state_co2, by = c("NTD_ID", "state")) |>
  mutate(
    electric_emissions = electric_mwh * coalesce(CO2_MWh, 785),
    total_transit_emissions_lbs = coalesce(fuel_emissions, 0) + coalesce(electric_emissions, 0),
    
    # Baseline driving emissions: ~0.80 lbs CO2 per passenger mile
    baseline_driving_emissions_lbs = sum_passenger_miles * 0.80,
    gta_vi_avoided_lbs = baseline_driving_emissions_lbs - total_transit_emissions_lbs
  ) |>
  group_by(NTD_ID, AGENCY, state) |>
  summarize(
    total_upt = sum(total_upt, na.rm = TRUE),
    total_passenger_miles = sum(sum_passenger_miles, na.rm = TRUE),
    total_transit_emissions = sum(total_transit_emissions_lbs, na.rm = TRUE),
    total_baseline_emissions = sum(baseline_driving_emissions_lbs, na.rm = TRUE),
    gta_vi = sum(gta_vi_avoided_lbs, na.rm = TRUE),
    .groups = "drop"
  ) |>
  filter(total_passenger_miles > 0) |>
  mutate(
    emissions_per_passenger_mile = total_transit_emissions / total_passenger_miles,
    agency_size = case_when(
      total_upt > 20000000 ~ "Large",
      total_upt > 2000000  ~ "Medium",
      TRUE                 ~ "Small"
    )
  )

Overall, the 2025 GTA Value Index shows that a transit agency’s environmental impact comes down to both ridership volume and the energy powering its fleet. High-ridership systems like MTA NYC Transit and long-distance commuter services like OmniRide and Short Line prove how much carbon transit can keep out of the atmosphere—while high-emission outliers show where cleaner fuel and grid upgrades are needed most.
